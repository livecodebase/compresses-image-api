const fs = require('fs');
const path = require('path');
const sharp = require('sharp');
const { execa } = require('execa');

const compressWithTool = async (tool, args) => {
  try {
    await execa(tool, args);
  } catch (err) {
    console.error(`[ERROR] ${tool}:`, err.stderr || err.message);
  }
};

const compressImage = async (filePath) => {
  const ext = path.extname(filePath).toLowerCase();
  const outputDir = "compressed";
  fs.mkdirSync(outputDir, { recursive: true });

  const fileName = path.basename(filePath);
  const outputPath = path.join(outputDir, fileName);

  // Fallback compression with sharp
  const compressWithSharp = async () => {
    await sharp(filePath)
      .toFormat("jpeg", { quality: 70 })
      .toFile(outputPath);
    console.log(`✅ Compressed with sharp: ${outputPath}`);
  };

  switch (ext) {
    case '.jpg':
    case '.jpeg':
      await compressWithTool('jpegtran', ['-copy', 'none', '-optimize', '-progressive', '-outfile', outputPath, filePath]);
      await compressWithTool('jpegoptim', ['--max=70', '--strip-all', outputPath]);
      break;

    case '.png':
      await compressWithTool('pngquant', ['--quality=60-80', '--output', outputPath, '--force', filePath]);
      await compressWithTool('optipng', ['-o7', outputPath]);
      await compressWithTool('pngcrush', ['-brute', outputPath, `${outputPath}_crush.png`]);
      break;

    case '.webp':
      const webpOut = path.join(outputDir, fileName.replace(ext, '.webp'));
      await compressWithTool('cwebp', ['-q', '70', filePath, '-o', webpOut]);
      break;

    case '.avif':
      const avifOut = path.join(outputDir, fileName.replace(ext, '.avif'));
      await compressWithTool('avifenc', ['--min', '30', '--max', '50', filePath, avifOut]);
      break;

    default:
      console.warn(`❓ Unsupported format ${ext}. Using sharp as fallback.`);
      await compressWithSharp();
  }
};

const run = async () => {
  const inputDir = 'images';
  const files = fs.readdirSync(inputDir);

  for (const file of files) {
    const filePath = path.join(inputDir, file);
    console.log(`🔄 Compressing: ${file}`);
    await compressImage(filePath);
  }
};

run();
